import {
  prisma,
} from "@/lib/prisma";

const VALIDATED_MODEL_VERSION =
  "holdout-v1-70-30";

export type SmartPickTier =
  | "VERY_HIGH"
  | "HIGH"
  | "MEDIUM"
  | "LOW";

  totalPicks: number;

  settledPicks: number;

  wins: number;
  losses: number;
  voids: number;

  hitRate: number;

  veryHighPicks: number;
  veryHighWins: number;
  veryHighLosses: number;
  veryHighHitRate: number;

  probability90Picks: number;
  probability90Wins: number;
  probability90Losses: number;
  probability90HitRate: number;

  leagues: number;

  matches: number;
};

export type SmartPickTierPerformance = {
  tier: SmartPickTier;

  picks: number;

  wins: number;
  losses: number;
  voids: number;

  settled: number;

  hitRate: number;

  averageProbability: number;

  averageReliability: number;
};

  key: string;

  label: string;

  minimum: number;
  maximum: number;

  picks: number;

  wins: number;
  losses: number;
  voids: number;

  settled: number;

  hitRate: number;

  averageProbability: number;

  averageReliability: number;

  calibrationGap: number;
};

export type SmartPickLeaguePerformance = {
  apiId: number;

  league: string;

  picks: number;

  wins: number;
  losses: number;
  voids: number;

  settled: number;

  hitRate: number;

  averageProbability: number;

  averageReliability: number;
};

export type SmartPickMarketPerformance = {
  key: string;

  market: string;
  selection: string;

  picks: number;

  wins: number;
  losses: number;
  voids: number;

  settled: number;

  hitRate: number;

  averageProbability: number;

  averageReliability: number;
};

  matchId: number;

  leagueName: string;

  marketKey: string;

  market: string;
  selection: string;

  probability: number;

  fairOdds: number | null;

};

export type SmartPicksPerformanceData = {

  summary:
    SmartPickPerformanceSummary;

  tiers:
    SmartPickTierPerformance[];

  probabilityBuckets:
    SmartPickProbabilityPerformance[];

  leagues:
    SmartPickLeaguePerformance[];

  markets:
    SmartPickMarketPerformance[];

  history:
    SmartPickHistoryRow[];
};

type PerformanceRow = {
  id: number;

  matchId: number;

  kickoffAt: Date;

  leagueApiId: number;
  leagueName: string;

  homeTeam: string;
  awayTeam: string;

  marketKey: string;

  market: string;
  selection: string;

  probability: number;

  fairOdds: number | null;

  reliabilityScore: number;
  pickScore: number;

  tier: string;

  actualHomeScore: number;
  actualAwayScore: number;

  result: string;

  rank: number;
};

type Accumulator = {
  picks: number;

  wins: number;
  losses: number;
  voids: number;

  probabilityTotal: number;

  reliabilityTotal: number;
};

const PROBABILITY_BUCKETS = [
  {
    key:
      "60_69",

    label:
      "60â€“69%",

    minimum:
      60,

    maximum:
      69.999,
  },

  {
    key:
      "70_79",

    label:
      "70â€“79%",

    minimum:
      70,

    maximum:
      79.999,
  },

  {
    key:
      "80_89",

    label:
      "80â€“89%",

    minimum:
      80,

    maximum:
      89.999,
  },

  {
    key:
      "90_100",

    label:
      "90%+",

    minimum:
      90,

    maximum:
      100,
  },
] as const;

const TIER_ORDER:
  SmartPickTier[] = [
    "VERY_HIGH",
    "HIGH",
    "MEDIUM",
    "LOW",
  ];

function round(
  value: number,
  decimals = 2,
): number {
  const factor =
    10 ** decimals;

  return (
    Math.round(
      value *
        factor,
    ) /
    factor
  );
}

function createAccumulator(): Accumulator {
  return {
    picks:
      0,

    wins:
      0,

    losses:
      0,

    voids:
      0,

    probabilityTotal:
      0,

    reliabilityTotal:
      0,
  };
}

function updateAccumulator(
  accumulator:
    Accumulator,

  row:
    PerformanceRow,
): void {
  accumulator.picks +=
    1;

  accumulator.probabilityTotal +=
    row.probability;

  accumulator.reliabilityTotal +=
    row.reliabilityScore;

  if (
    row.result ===
    "WIN"
  ) {
    accumulator.wins +=
      1;

    return;
  }

  if (
    row.result ===
    "LOSS"
  ) {
    accumulator.losses +=
      1;

    return;
  }

  accumulator.voids +=
    1;
}

function calculateSettled(
  accumulator:
    Accumulator,
): number {
  return (
    accumulator.wins +
    accumulator.losses
  );
}

function calculateHitRate(
  accumulator:
    Accumulator,
): number {
  const settled =
    calculateSettled(
      accumulator,
    );

  if (
    settled ===
    0
  ) {
    return 0;
  }

  return round(
    (
      accumulator.wins /
      settled
    ) *
      100,
  );
}

function calculateAverageProbability(
  accumulator:
    Accumulator,
): number {
  if (
    accumulator.picks ===
    0
  ) {
    return 0;
  }

  return round(
    accumulator.probabilityTotal /
      accumulator.picks,
  );
}

function calculateAverageReliability(
  accumulator:
    Accumulator,
): number {
  if (
    accumulator.picks ===
    0
  ) {
    return 0;
  }

  return round(
    accumulator.reliabilityTotal /
      accumulator.picks,
  );
}

function normalizeTier(
  tier: string,
): SmartPickTier {
  if (
    tier === "VERY_HIGH" ||
    tier === "HIGH" ||
    tier === "MEDIUM" ||
    tier === "LOW"
  ) {
    return tier;
  }

  return "LOW";
}

export async function loadSmartPicksPerformance(): Promise<SmartPicksPerformanceData> {
  const rows =
    await prisma.smartPickHistory.findMany({
      where: {
        modelVersion:
          VALIDATED_MODEL_VERSION,
      },

      select: {
        id:
          true,

        matchId:
          true,

        kickoffAt:
          true,

        leagueApiId:
          true,

        leagueName:
          true,

        homeTeam:
          true,

        awayTeam:
          true,

        marketKey:
          true,

        market:
          true,

        selection:
          true,

        probability:
          true,

        fairOdds:
          true,

        reliabilityScore:
          true,

        pickScore:
          true,

        tier:
          true,

        actualHomeScore:
          true,

        actualAwayScore:
          true,

        result:
          true,

        rank:
          true,
      },

      orderBy: [
        {
          kickoffAt:
            "desc",
        },

        {
          rank:
            "asc",
        },
      ],
    });

  const performanceRows:
    PerformanceRow[] =
      rows;

  /*
   * =========================================================
   * GLOBAL
   * =========================================================
   */

  const global =
    createAccumulator();

  const veryHigh =
    createAccumulator();

  const probability90 =
    createAccumulator();

  const matchIds =
    new Set<
      number
    >();

  const leagueIds =
    new Set<
      number
    >();

  for (
    const row
    of performanceRows
  ) {
    updateAccumulator(
      global,
      row,
    );

    matchIds.add(
      row.matchId,
    );

    leagueIds.add(
      row.leagueApiId,
    );

    if (
      row.tier ===
      "VERY_HIGH"
    ) {
      updateAccumulator(
        veryHigh,
        row,
      );
    }

    if (
      row.probability >=
      90
    ) {
      updateAccumulator(
        probability90,
        row,
      );
    }
  }

  const summary:
    SmartPickPerformanceSummary = {
      totalPicks:
        global.picks,

      settledPicks:
        calculateSettled(
          global,
        ),

      wins:
        global.wins,

      losses:
        global.losses,

      voids:
        global.voids,

      hitRate:
        calculateHitRate(
          global,
        ),

      veryHighPicks:
        veryHigh.picks,

      veryHighWins:
        veryHigh.wins,

      veryHighLosses:
        veryHigh.losses,

      veryHighHitRate:
        calculateHitRate(
          veryHigh,
        ),

      probability90Picks:
        probability90.picks,

      probability90Wins:
        probability90.wins,

      probability90Losses:
        probability90.losses,

      probability90HitRate:
        calculateHitRate(
          probability90,
        ),

      leagues:
        leagueIds.size,

      matches:
        matchIds.size,
    };

  /*
   * =========================================================
   * TIERS
   * =========================================================
   */

  const tierMap =
    new Map<
      SmartPickTier,
      Accumulator
    >();

  for (
    const tier
    of TIER_ORDER
  ) {
    tierMap.set(
      tier,
      createAccumulator(),
    );
  }

  for (
    const row
    of performanceRows
  ) {
    const tier =
      normalizeTier(
        row.tier,
      );

    const accumulator =
      tierMap.get(
        tier,
      );

    if (
      accumulator
    ) {
      updateAccumulator(
        accumulator,
        row,
      );
    }
  }

  const tiers:
    SmartPickTierPerformance[] =
      TIER_ORDER.map(
        (
          tier,
        ) => {
          const accumulator =
            tierMap.get(
              tier,
            ) ??
            createAccumulator();

          return {
            tier,

            picks:
              accumulator.picks,

            wins:
              accumulator.wins,

            losses:
              accumulator.losses,

            voids:
              accumulator.voids,

            settled:
              calculateSettled(
                accumulator,
              ),

            hitRate:
              calculateHitRate(
                accumulator,
              ),

            averageProbability:
              calculateAverageProbability(
                accumulator,
              ),

            averageReliability:
              calculateAverageReliability(
                accumulator,
              ),
          };
        },
      );

  /*
   * =========================================================
   * PROBABILITY BUCKETS
   * =========================================================
   */

  const probabilityBuckets:
    SmartPickProbabilityPerformance[] =
      PROBABILITY_BUCKETS.map(
        (
          bucket,
        ) => {
          const accumulator =
            createAccumulator();

          for (
            const row
            of performanceRows
          ) {
            if (
              row.probability <
                bucket.minimum ||
              row.probability >
                bucket.maximum
            ) {
              continue;
            }

            updateAccumulator(
              accumulator,
              row,
            );
          }

          const averageProbability =
            calculateAverageProbability(
              accumulator,
            );

          const actualHitRate =
            calculateHitRate(
              accumulator,
            );

          return {
            key:
              bucket.key,

            label:
              bucket.label,

            minimum:
              bucket.minimum,

            maximum:
              bucket.maximum,

            picks:
              accumulator.picks,

            wins:
              accumulator.wins,

            losses:
              accumulator.losses,

            voids:
              accumulator.voids,

            settled:
              calculateSettled(
                accumulator,
              ),

            hitRate:
              actualHitRate,

            averageProbability,

            averageReliability:
              calculateAverageReliability(
                accumulator,
              ),

            calibrationGap:
              round(
                actualHitRate -
                  averageProbability,
              ),
          };
        },
      );

  /*
   * =========================================================
   * LEAGUES
   * =========================================================
   */

  const leagueMap =
    new Map<
      number,
      {
        league:
          string;

        accumulator:
          Accumulator;
      }
    >();

  for (
    const row
    of performanceRows
  ) {
    let league =
      leagueMap.get(
        row.leagueApiId,
      );

    if (
      !league
    ) {
      league = {
        league:
          row.leagueName,

        accumulator:
          createAccumulator(),
      };

      leagueMap.set(
        row.leagueApiId,
        league,
      );
    }

    updateAccumulator(
      league.accumulator,
      row,
    );
  }

  const leagues:
    SmartPickLeaguePerformance[] =
      [
        ...leagueMap.entries(),
      ]
        .map(
          (
            [
              apiId,
              value,
            ],
          ) => ({
            apiId,

            league:
              value.league,

            picks:
              value
                .accumulator
                .picks,

            wins:
              value
                .accumulator
                .wins,

            losses:
              value
                .accumulator
                .losses,

            voids:
              value
                .accumulator
                .voids,

            settled:
              calculateSettled(
                value.accumulator,
              ),

            hitRate:
              calculateHitRate(
                value.accumulator,
              ),

            averageProbability:
              calculateAverageProbability(
                value.accumulator,
              ),

            averageReliability:
              calculateAverageReliability(
                value.accumulator,
              ),
          }),
        )
        .sort(
          (
            first,
            second,
          ) =>
            second.hitRate -
            first.hitRate,
        );

  /*
   * =========================================================
   * MARKETS
   * =========================================================
   */

  const marketMap =
    new Map<
      string,
      {
        market:
          string;

        selection:
          string;

        accumulator:
          Accumulator;
      }
    >();

  for (
    const row
    of performanceRows
  ) {
    let market =
      marketMap.get(
        row.marketKey,
      );

    if (
      !market
    ) {
      market = {
        market:
          row.market,

        selection:
          row.selection,

        accumulator:
          createAccumulator(),
      };

      marketMap.set(
        row.marketKey,
        market,
      );
    }

    updateAccumulator(
      market.accumulator,
      row,
    );
  }

  const markets:
    SmartPickMarketPerformance[] =
      [
        ...marketMap.entries(),
      ]
        .map(
          (
            [
              key,
              value,
            ],
          ) => ({
            key,

            market:
              value.market,

            selection:
              value.selection,

            picks:
              value
                .accumulator
                .picks,

            wins:
              value
                .accumulator
                .wins,

            losses:
              value
                .accumulator
                .losses,

            voids:
              value
                .accumulator
                .voids,

            settled:
              calculateSettled(
                value.accumulator,
              ),

            hitRate:
              calculateHitRate(
                value.accumulator,
              ),

            averageProbability:
              calculateAverageProbability(
                value.accumulator,
              ),

            averageReliability:
              calculateAverageReliability(
                value.accumulator,
              ),
          }),
        )
        .filter(
          (
            market,
          ) =>
            market.picks >=
            20,
        )
        .sort(
          (
            first,
            second,
          ) => {
            const hitRateDifference =
              second.hitRate -
              first.hitRate;

            if (
              Math.abs(
                hitRateDifference,
              ) >
              0.001
            ) {
              return hitRateDifference;
            }

            return (
              second.picks -
              first.picks
            );
          },
        );

  /*
   * =========================================================
   * HISTORY
   * =========================================================
   */

  const history:
    SmartPickHistoryRow[] =
      performanceRows.slice(
        0,
        200,
      );

  return {
    modelVersion:
      VALIDATED_MODEL_VERSION,

    summary,

    tiers,

    probabilityBuckets,

    leagues,

    markets,

    history,
  };
}
