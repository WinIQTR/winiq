function getCalibrationClass(

function round(
  value: number,
  decimals = 2,
): number {
  const factor =
    10 ** decimals;

  return (
    Math.round(
      value *
        factor,
    ) /
    factor
  );
}

function resolveCompetitionApiId(
  rawValue:
    | string
    | string[]
    | undefined,
): number {
  const firstActiveCompetition =
    ACTIVE_COMPETITIONS[0];

  if (
    !firstActiveCompetition
  ) {
    throw new Error(
      "Aktif organizasyon bulunamadÄ±.",
    );
  }

  const normalizedValue =
    Array.isArray(
      rawValue,
    )
      ? rawValue[0]
      : rawValue;

  if (
    !normalizedValue
  ) {
    return firstActiveCompetition.apiId;
  }

  const parsedValue =
    Number.parseInt(
      normalizedValue,
      10,
    );

  if (
    !Number.isInteger(
      parsedValue,
    )
  ) {
    return firstActiveCompetition.apiId;
  }

  const competition =
    getCompetitionByApiId(
      parsedValue,
    );

  if (
    !competition ||
    !competition.enabled
  ) {
    return firstActiveCompetition.apiId;
  }

  return competition.apiId;
}

function getPositionZone(
  options: {
    position: number;
    teamCount: number;
    competition:

  if (

}
