function round(
  value: number,
  decimals = 2,
): number {
  const factor =
    10 ** decimals;

  return (
    Math.round(
      value * factor,
    ) / factor
  );
}

function calculatePercentage(
  numerator: number,
  denominator: number,
): number {
  if (denominator <= 0) {
    return 0;
  }

  return round(
    (
      numerator /
      denominator
    ) * 100,
  );
}

function getActualOutcome(
  homeScore: number,
  awayScore: number,
): ActualOutcome {
  if (
    homeScore >
    awayScore
  ) {
    return "HOME";
  }

  if (
    awayScore >
    homeScore
  ) {
    return "AWAY";
  }

  return "DRAW";
}

function getActualProbability(
  prediction: DashboardPrediction,
  actualOutcome: ActualOutcome,
): number {
  if (
    actualOutcome ===
    "HOME"
  ) {
    return prediction.homeProbability;
  }

  if (
    actualOutcome ===
    "DRAW"
  ) {
    return prediction.drawProbability;
  }

  return prediction.awayProbability;
}

function buildProbabilityBuckets(
  predictions: EvaluatedPrediction[],
): ProbabilityBucket[] {
  const definitions = [
    {
      label: "0-39.9%",
      min: 0,
      max: 40,
    },
    {
      label: "40-44.9%",
      min: 40,
      max: 45,
    },
    {
      label: "45-49.9%",
      min: 45,
      max: 50,
    },
    {
      label: "50-54.9%",
      min: 50,
      max: 55,
    },
    {
      label: "55-59.9%",
      min: 55,
      max: 60,
    },
    {
      label: "60-64.9%",
      min: 60,
      max: 65,
    },
    {
      label: "65%+",
      min: 65,
      max: null,
    },
  ];

  return definitions.map(
    (definition) => {
      const items =
        predictions.filter(
          (prediction) => {
            const probability =
              prediction.predictedProbability;

            if (
              probability <
              definition.min
            ) {
              return false;
            }

            if (
              definition.max !==
              null &&
              probability >=
                definition.max
            ) {
              return false;
            }

            return true;
          },
        );

      const correct =
        items.filter(
          (prediction) =>
            prediction.isCorrect,
        ).length;

      const averagePredictedProbability =
        items.length > 0
          ? round(
              items.reduce(
                (
                  total,
                  prediction,
                ) =>
                  total +
                  prediction
                    .predictedProbability,
                0,
              ) /
                items.length,
            )
          : 0;

      const actualAccuracy =
        calculatePercentage(
          correct,
          items.length,
        );

      return {
        label:
          definition.label,

        minProbability:
          definition.min,

        maxProbability:
          definition.max,

        totalPredictions:
          items.length,

        correctPredictions:
          correct,

        incorrectPredictions:
          items.length -
          correct,

        averagePredictedProbability,

        actualAccuracy,

        calibrationGap:
          round(
            actualAccuracy -
              averagePredictedProbability,
          ),
      };
    },
  );
}

export async function evaluatePredictions(
  limit = 100,
): Promise<PredictionEvaluationResult> {
  const predictions =
    await loadDashboardPredictions(
      limit,
    );

  if (
    predictions.length === 0
  ) {
    return {
      summary:
        createEmptySummary(),

      predictions: [],
    };
  }

  const matchIds =
    predictions.map(
      (prediction) =>
        prediction.matchId,
    );

  const matches =
    await prisma.match.findMany({
      where: {
        id: {
          in: matchIds,
        },

        homeScore: {
          not: null,
        },

        awayScore: {
          not: null,
        },
      },

      select: {
        id: true,
        homeScore: true,
        awayScore: true,
      },
    });

  const resultByMatchId =
    new Map(
      matches.map(
        (match) => [
          match.id,
          match,
        ],
      ),
    );

  const evaluated:
    EvaluatedPrediction[] = [];

  for (
    const prediction
    of predictions
  ) {
    const match =
      resultByMatchId.get(
        prediction.matchId,
      );

    if (
      !match ||
      match.homeScore === null ||
      match.awayScore === null
    ) {
      continue;
    }

    const actualOutcome =
      getActualOutcome(
        match.homeScore,
        match.awayScore,
      );

    const isCorrect =
      prediction.predictedOutcome ===
      actualOutcome;

    const actualProbability =
      getActualProbability(
        prediction,
        actualOutcome,
      );

    const probabilityError =
      round(
        100 -
          actualProbability,
      );

    evaluated.push({
      ...prediction,

      actualHomeScore:
        match.homeScore,

      actualAwayScore:
        match.awayScore,

      actualOutcome,

      isCorrect,

      probabilityError,
    });
  }

  const correctPredictions =
    evaluated.filter(
      (prediction) =>
        prediction.isCorrect,
    ).length;

  const home =
    evaluated.filter(
      (prediction) =>
        prediction.predictedOutcome ===
        "HOME",
    );

  const draw =
    evaluated.filter(
      (prediction) =>
        prediction.predictedOutcome ===
        "DRAW",
    );

  const away =
    evaluated.filter(
      (prediction) =>
        prediction.predictedOutcome ===
        "AWAY",
    );

  const highConfidence =
    evaluated.filter(
      (prediction) =>
        prediction.confidenceScore >=
        70,
    );

  const mediumConfidence =
    evaluated.filter(
      (prediction) =>
        prediction.confidenceScore >=
          50 &&
        prediction.confidenceScore <
          70,
    );

  const lowConfidence =
    evaluated.filter(
      (prediction) =>
        prediction.confidenceScore <
        50,
    );

  const countCorrect = (
    items:
      EvaluatedPrediction[],
  ): number =>
    items.filter(
      (prediction) =>
        prediction.isCorrect,
    ).length;

  const averageWinningProbability =
    evaluated.length > 0
      ? round(
          evaluated.reduce(
            (
              total,
              prediction,
            ) =>
              total +
              prediction
                .predictedProbability,
            0,
          ) /
            evaluated.length,
        )
      : 0;

  const summary:
    PredictionEvaluationSummary = {
      totalPredictions:
        evaluated.length,

      correctPredictions,

      incorrectPredictions:
        evaluated.length -
        correctPredictions,

      accuracy:
        calculatePercentage(
          correctPredictions,
          evaluated.length,
        ),

      homePredictions:
        home.length,

      homeCorrect:
        countCorrect(home),

      homeAccuracy:
        calculatePercentage(
          countCorrect(home),
          home.length,
        ),

      drawPredictions:
        draw.length,

      drawCorrect:
        countCorrect(draw),

      drawAccuracy:
        calculatePercentage(
          countCorrect(draw),
          draw.length,
        ),

      awayPredictions:
        away.length,

      awayCorrect:
        countCorrect(away),

      awayAccuracy:
        calculatePercentage(
          countCorrect(away),
          away.length,
        ),

      highConfidencePredictions:
        highConfidence.length,

      highConfidenceCorrect:
        countCorrect(
          highConfidence,
        ),

      highConfidenceAccuracy:
        calculatePercentage(
          countCorrect(
            highConfidence,
          ),
          highConfidence.length,
        ),

      mediumConfidencePredictions:
        mediumConfidence.length,

      mediumConfidenceCorrect:
        countCorrect(
          mediumConfidence,
        ),

      mediumConfidenceAccuracy:
        calculatePercentage(
          countCorrect(
            mediumConfidence,
          ),
          mediumConfidence.length,
        ),

      lowConfidencePredictions:
        lowConfidence.length,

      lowConfidenceCorrect:
        countCorrect(
          lowConfidence,
        ),

      lowConfidenceAccuracy:
        calculatePercentage(
          countCorrect(
            lowConfidence,
          ),
          lowConfidence.length,
        ),

      averageWinningProbability,

      probabilityBuckets:
        buildProbabilityBuckets(
          evaluated,
        ),

      confidenceBuckets:
        buildConfidenceBuckets(
          evaluated,
        ),
    };

  return {
    summary,

    predictions:
      evaluated,
  };
}
